FONCTION ConvertDecimalToBinary(nombre)
	CALCULE POUR TRANSFORMER nombre EN binaire
		UTILISER MODULO
	FIN CALCULE
FIN FONCTION



FONCTION ConvertBinaryToDecimal(nombre)
	CALCULE POUR TRANSFORMER nombre EN decimal
		UTILISER INVERSE DE MODULO (multiplier)
	FIN CALCULE
FIN FONCTION



FONCTION ConvertBinaryToOctal(nombre)
	CALCULE POUR TRANSFORMER nombre EN octal
		REGROUPER BITS PAR GROUPE DE 3 PUIS TRANSFORMER EN DECIMAL
		(exemple: 101 -> 1x4 + 0x2 + 1x1 = 5
		101 -> 1x4 + 1x2 + 0x1 = 6
		résultat : 56 (faire une table de correspondance))
	FIN CALCULE
/*

*/
FIN FONCTION
